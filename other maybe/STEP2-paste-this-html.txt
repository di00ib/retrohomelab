// ===== FEEDBACK WIDGET =====
// Replace YOUR_FORM_ID with your actual Formspree form ID (e.g. 'xabc1234')
const FORMSPREE_ID = 'YOUR_FORM_ID';

function openFeedback() {
  document.getElementById('fbForm').style.display = '';
  document.getElementById('fbSuccess').style.display = 'none';
  document.getElementById('fbError').style.display = 'none';
  document.getElementById('fbSubmit').disabled = false;
  document.getElementById('fbSubmitLabel').textContent = 'Send Message';
  document.getElementById('fbOverlay').classList.add('open');
  document.body.style.overflow = 'hidden';
  setTimeout(() => document.getElementById('fb-name').focus(), 300);
}

function closeFeedback() {
  document.getElementById('fbOverlay').classList.remove('open');
  document.body.style.overflow = '';
}

document.addEventListener('keydown', function(e) {
  if (e.key === 'Escape') closeFeedback();
});

async function submitFeedback() {
  const name    = document.getElementById('fb-name').value.trim();
  const email   = document.getElementById('fb-email').value.trim();
  const phone   = document.getElementById('fb-phone').value.trim();
  const message = document.getElementById('fb-message').value.trim();
  const type    = document.getElementById('fb-type').value;
  const errEl   = document.getElementById('fbError');
  const btn     = document.getElementById('fbSubmit');
  const lbl     = document.getElementById('fbSubmitLabel');

  errEl.style.display = 'none';
  if (!name)                          { showFbError('Please enter your name.'); return; }
  if (!email || !email.includes('@')) { showFbError('Please enter a valid email address.'); return; }
  if (!message)                       { showFbError('Please write a message before sending.'); return; }

  btn.disabled = true;
  lbl.textContent = 'Sending\u2026';

  try {
    const res = await fetch('https://formspree.io/f/' + FORMSPREE_ID, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
      body: JSON.stringify({
        type:     type,
        name:     name,
        email:    email,
        phone:    phone || 'Not provided',
        message:  message,
        page:     window.location.hostname + window.location.pathname,
        _replyto: email,
        _subject: '[Retro Homelab] ' + type + ' from ' + name
      })
    });

    if (res.ok) {
      document.getElementById('fbForm').style.display = 'none';
      document.getElementById('fbSuccess').style.display = 'block';
    } else {
      const data = await res.json().catch(() => ({}));
      showFbError(data.error || 'Something went wrong. Please email donreynolds+heartjournal@gmail.com directly.');
      btn.disabled = false;
      lbl.textContent = 'Send Message';
    }
  } catch (err) {
    showFbError('Could not send \u2014 check your internet connection, or email donreynolds+heartjournal@gmail.com directly.');
    btn.disabled = false;
    lbl.textContent = 'Send Message';
  }
}

function showFbError(msg) {
  const el = document.getElementById('fbError');
  el.textContent = msg;
  el.style.display = 'block';
  el.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
}
</script>

============================================================
FORMSPREE SETUP — takes about 2 minutes
============================================================

1. Go to https://formspree.io and click "Get Started Free"
2. Sign up with donreynolds@gmail.com
3. Click "New Form"
4. Name it: "Retro Homelab Feedback"
5. Set the email to: donreynolds+heartjournal@gmail.com
6. Click "Create Form"
7. You'll get a URL like: https://formspree.io/f/abcd1234
   Your form ID is the part after /f/ — in this example: abcd1234
8. In the heartjournal/index.html file, find:
      const FORMSPREE_ID = 'YOUR_FORM_ID';
   Replace YOUR_FORM_ID with your actual ID (keep the quotes):
      const FORMSPREE_ID = 'abcd1234';
9. Do the same in the landing page index.html
10. Save and commit — done!

First submission will trigger a confirmation email to donreynolds@gmail.com.
Click the confirmation link and you're live.

Free tier: 50 submissions/month. More than enough for a hobby project.
